\section*{Summary of Changes}

\textbf{All programs}
\begin{itemize}
\item PHY 9A or 9HA, followed by the remainder of one series (9B--9D or 9HB--9HD);  PHY 9HE removed.
\item MAT 22A and 22B at 4 units, with MAT/BIS 27A or MAT 67, and MAT/BIS 27B, as alternatives.
\item PHY 198 and 199 must be taken for a total of at least 3 units.
\item PHY 185 and 190 recommended.
\end{itemize}

\textbf{BS Physics}
\begin{itemize}
\item PHY 45 added;  PHY 110C and PHY 102/104B removed.
\item Laboratory:  PHY 122A, 122B, or both PHY 117 and 118, in place of the 116 series.
\item A total of 14 upper division Physics courses of 3 or more units (from 15).
\item Minimum 108 units.
\end{itemize}

\textbf{BS Physics, Astrophysics Emphasis}
\begin{itemize}
\item Four of PHY 151, 152, 153, 156, 158 (PHY 158 added);  PHY 102/104B removed.
\item Electives:  PHY 117 and 118 added;  PHY 110C, 116A, and GEL 163 removed.
\item Minimum 109 units.
\end{itemize}

\textbf{AB Physics}
\begin{itemize}
\item PHY 40 added;  PHY 102 removed.
\item PHY 110B moved from the required courses to the electives, with PHY 156 and 158 added.
\item Laboratory:  one of PHY 117, 118, 122A, 122B (122A and 122B require PHY 110B).
\item The additional fixed-unit upper division course removed.
\item Minimum 78 units.
\end{itemize}
